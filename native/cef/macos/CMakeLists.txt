# Included from the unmodified CEF SDK's root build so its wrapper, sandbox,
# framework loader and helper layout use the vendor's platform configuration.
if(NOT OS_MAC)
  message(FATAL_ERROR "The macOS browser bridge requires macOS")
endif()
set(CEF_TARGET QQClawBrowser)
set(CEF_HELPER_TARGET QQClawBrowser_Helper)
set(CEF_HELPER_OUTPUT_NAME "QQClawBrowser Helper")
SET_CEF_TARGET_OUT_DIR()
set(CEF_APP "${CEF_TARGET_OUT_DIR}/${CEF_TARGET}.app")
set(EXECUTABLE_NAME "${CEF_TARGET}")
set(PRODUCT_NAME "${CEF_TARGET}")
set(BUNDLE_ID_SUFFIX "")

add_library(QQClawCefBridge SHARED CefBridge.mm)
SET_LIBRARY_TARGET_PROPERTIES(QQClawCefBridge)
target_compile_options(QQClawCefBridge PRIVATE -fobjc-arc)
target_compile_definitions(QQClawCefBridge PRIVATE CEF_API_VERSION=15400)
target_link_libraries(QQClawCefBridge PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS} "-framework WebKit" "-framework QuartzCore")
set_target_properties(QQClawCefBridge PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE)

add_executable(${CEF_TARGET} MACOSX_BUNDLE "${QQCLAW_CEF_ENTRY}")
SET_EXECUTABLE_TARGET_PROPERTIES(${CEF_TARGET})
target_include_directories(${CEF_TARGET} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(${CEF_TARGET} PRIVATE -fobjc-arc)
target_link_libraries(${CEF_TARGET} PRIVATE QQClawCefBridge ${CEF_STANDARD_LIBS} "-framework WebKit")
set_target_properties(${CEF_TARGET} PROPERTIES
  MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in"
  BUILD_RPATH "@executable_path/../Frameworks")
COPY_MAC_FRAMEWORK("${CEF_TARGET}" "${CEF_BINARY_DIR}" "${CEF_APP}")
add_custom_command(TARGET ${CEF_TARGET} POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:QQClawCefBridge>" "${CEF_APP}/Contents/Frameworks/libQQClawCefBridge.dylib")

foreach(_suffix_list ${CEF_HELPER_APP_SUFFIXES})
  string(REPLACE ":" ";" _suffix_list "${_suffix_list}")
  list(GET _suffix_list 0 _name_suffix)
  list(GET _suffix_list 1 _target_suffix)
  list(GET _suffix_list 2 _plist_suffix)
  set(_helper_target "${CEF_HELPER_TARGET}${_target_suffix}")
  set(_helper_name "${CEF_HELPER_OUTPUT_NAME}${_name_suffix}")
  set(_helper_plist "${CMAKE_CURRENT_BINARY_DIR}/helper${_target_suffix}.plist")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in" _plist)
  string(REPLACE "\${EXECUTABLE_NAME}" "${_helper_name}" _plist "${_plist}")
  string(REPLACE "\${PRODUCT_NAME}" "${_helper_name}" _plist "${_plist}")
  string(REPLACE "\${BUNDLE_ID_SUFFIX}" ".helper${_plist_suffix}" _plist "${_plist}")
  file(WRITE "${_helper_plist}" "${_plist}")
  add_executable(${_helper_target} MACOSX_BUNDLE Helper.cc)
  SET_EXECUTABLE_TARGET_PROPERTIES(${_helper_target})
  target_link_libraries(${_helper_target} PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
  set_target_properties(${_helper_target} PROPERTIES OUTPUT_NAME "${_helper_name}" MACOSX_BUNDLE_INFO_PLIST "${_helper_plist}")
  add_dependencies(${CEF_TARGET} ${_helper_target})
  add_custom_command(TARGET ${CEF_TARGET} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_TARGET_OUT_DIR}/${_helper_name}.app" "${CEF_APP}/Contents/Frameworks/${_helper_name}.app")
endforeach()
